📦 Sistema Full-Stack: Controlo de Stock (Guia Definitivo SAEP - Arquitetura Limpa e CRUD Completo)

Este material detalha a construção estruturada do sistema de controlo de stock, garantindo que todas as entidades (Utilizador, Produto e Movimentação) possuam o ciclo CRUD completo com as suas responsabilidades rigidamente separadas. Este tutorial contém 100% do código e documentação necessários para a prova prática.

🛠️ Fase 0: Preparação do Ambiente e Instalação

1. Criar o projeto base (Next.js):

Abra o terminal e gere a estrutura inicial:

npx create-next-app@latest controlo-stock
cd controlo-stock

2. Instalar as dependências do Frontend e Backend:

npm install axios sweetalert2
npm install @prisma/client@5
npm install prisma@5 --save-dev

3. Inicializar o Prisma:

npx prisma init

model Usuario {

  nome     String @db.VarChar(255)

  movimentacoes Movimentacao[]

model Produto {
  id             Int     @id @default(autoincrement())
  codigo         String  @unique @db.VarChar(50)
  descricao      String  @db.VarChar(255)
  quantidade     Int     @default(0)
  valor_unitario Decimal @db.Decimal(10,2)
  estoque_minimo Int     @default(10)
  movimentacoes Movimentacao[]
  @@map("Produto")
}

model Movimentacao {
  id                Int      @id @default(autoincrement())
  produto_id        Int
  usuario_id        Int
  tipo              TipoMovimentacao
  quantidade        Int
  data_movimentacao DateTime @default(now())
  produto Produto @relation(fields: [produto_id], references: [id])
  usuario Usuario @relation(fields: [usuario_id], references: [id])
  @@map("Movimentacao")
}

enum TipoMovimentacao {
  ENTRADA
  SAIDA
}

3. Documentação SAEP (Entregas 1, 8 e 9)

Entrega 1: Requisitos Funcionais

  • RF01 - Autenticação: O sistema deve permitir que o utilizador inicie sessão com e-mail e palavra-passe.
  • RF02 - Gestão de Produtos: O sistema deve permitir registar, listar, procurar, atualizar e eliminar produtos.
  • RF03 - Movimentações: O sistema deve registar entradas e saídas de stock, atualizando o saldo automaticamente.
  • RF04 - Alerta de Stock: O sistema deve emitir um aviso visual caso uma saída deixe a quantidade igual ou inferior ao limite mínimo.

src/repositories/UsuarioRepository.ts

import prisma from '@/src/lib/prisma';
import { UsuarioEntity } from '../entities/UsuarioEntity';
import { UsuarioProps } from '../interfaces';

export class UsuarioRepository {
    async listarTodos(): Promise<UsuarioEntity[]> {
        const users = await prisma.usuario.findMany();
        return users.map(u => new UsuarioEntity({ id: u.id, nome: u.nome, email: u.email, senha: u.senha }));
    }

    async buscarPorId(id: number): Promise<UsuarioEntity | null> {
        const u = await prisma.usuario.findUnique({ where: { id } });
        return u ? new UsuarioEntity({ id: u.id, nome: u.nome, email: u.email, senha: u.senha }) : null;
    }

    async buscarPorEmail(email: string): Promise<UsuarioEntity | null> {
        const u = await prisma.usuario.findUnique({ where: { email } });
        return u ? new UsuarioEntity({ id: u.id, nome: u.nome, email: u.email, senha: u.senha }) : null;
    }

    async criar(props: UsuarioProps): Promise<UsuarioEntity> {
        const u = await prisma.usuario.create({
            data: { nome: props.nome, email: props.email, senha: props.senha! }
        });
        return new UsuarioEntity({ id: u.id, nome: u.nome, email: u.email, senha: u.senha });
    }

    async atualizar(id: number, props: Partial<UsuarioProps>): Promise<UsuarioEntity> {
        const u = await prisma.usuario.update({
            where: { id },
            data: { nome: props.nome, email: props.email, senha: props.senha }
        });
        return new UsuarioEntity({ id: u.id, nome: u.nome, email: u.email, senha: u.senha });
    }

    async excluir(id: number): Promise<void> {
        await prisma.usuario.delete({ where: { id } });
    }
}

src/services/UsuarioService.ts

import { UsuarioRepository } from '../repositories/UsuarioRepository';
import { UsuarioProps } from '../interfaces';

export class UsuarioService {
    constructor(private repo: UsuarioRepository) {}

    async autenticar(email: string, senhaLimpa: string) {
        const user = await this.repo.buscarPorEmail(email);
        if (!user || user.getSenha !== senhaLimpa) {
            throw new Error("Credenciais inválidas.");
        }
        return { id: user.getId, nome: user.getNome, email: user.getEmail };
    }

    async listar() {
        const usuarios = await this.repo.listarTodos();
        return usuarios.map(u => ({ id: u.getId, nome: u.getNome, email: u.getEmail }));
    }

    async buscar(id: number) {
        const u = await this.repo.buscarPorId(id);
        if (!u) throw new Error("Usuário não encontrado.");
        return { id: u.getId, nome: u.getNome, email: u.getEmail };
    }

    async cadastrar(props: UsuarioProps) {
        const u = await this.repo.criar(props);
        return { id: u.getId, nome: u.getNome, email: u.getEmail };
    }

    async atualizar(id: number, props: Partial<UsuarioProps>) {
        const u = await this.repo.atualizar(id, props);
        return { id: u.getId, nome: u.getNome, email: u.getEmail };
    }

    async excluir(id: number) {
        return await this.repo.excluir(id);
    }
}

src/services/ProdutoService.ts

import { ProdutoRepository } from '../repositories/ProdutoRepository';
import { ProdutoProps } from '../interfaces';

export class ProdutoService {
    constructor(private repo: ProdutoRepository) {}

    private mapper(p: any) {
        return {
            id: p.getId,
            codigo: p.getCodigo,
            descricao: p.getDescricao,
            quantidade: p.getQuantidade,
            valorUnitario: p.getValorUnitario,
            estoqueMinimo: p.getEstoqueMinimo
        };
    }

    async listar() {
        const lista = await this.repo.listarTodos();
        return lista.map(p => this.mapper(p));
    }

    async buscar(id: number) {
        const p = await this.repo.buscarPorId(id);
        if (!p) throw new Error("Produto não encontrado.");
        return this.mapper(p);
    }

    async cadastrar(props: ProdutoProps) {
        if (!props.codigo || !props.descricao) throw new Error("Código e Descrição são obrigatórios.");
        const p = await this.repo.criar(props);
        return this.mapper(p);
    }

    async atualizar(id: number, props: Partial<ProdutoProps>) {
        await this.buscar(id);
        const p = await this.repo.atualizar(id, props);
        return this.mapper(p);
    }

    async excluir(id: number) {
        await this.buscar(id);
        return await this.repo.excluir(id);
    }
}

src/services/MovimentacaoService.ts

import { MovimentacaoRepository } from '../repositories/MovimentacaoRepository';
import { ProdutoRepository } from '../repositories/ProdutoRepository';
import { MovimentacaoProps } from '../interfaces';

export class MovimentacaoService {
    constructor(
        private movRepo: MovimentacaoRepository,
        private prodRepo: ProdutoRepository
    ) {}

    async registrar(props: MovimentacaoProps) {
        if (props.quantidade <= 0) throw new Error("Quantidade deve ser maior que zero.");

        const produto = await this.prodRepo.buscarPorId(props.produtoId);
        if (!produto) throw new Error("Produto não encontrado.");

        if (props.tipo === 'SAIDA' && produto.getQuantidade < props.quantidade) {
            throw new Error("Estoque insuficiente.");
        }

        const novoEstoque = props.tipo === 'ENTRADA' 
            ? produto.getQuantidade + props.quantidade 
            : produto.getQuantidade - props.quantidade;

        // Atualização utilizando o setter encapsulado na entidade
        produto.setQuantidade = novoEstoque;

        await this.prodRepo.atualizar(props.produtoId, { quantidade: novoEstoque });
        await this.movRepo.criar(props);

        if (props.tipo === 'SAIDA' && produto.isEstoqueBaixo()) {
            return { sucesso: true, alerta: `Atenção: O estoque do produto ${produto.getDescricao()} atingiu o limite mínimo!` };
        }

        return { sucesso: true, alerta: null };
    }

    async listar() {
        const lista = await this.movRepo.listarTodos();
        return lista.map(m => ({
            id: m.getId,
            produtoId: m.getProdutoId,
            usuarioId: m.getUsuarioId,
            tipo: m.getTipo,
            quantidade: m.getQuantidade,
            data: m.getData
        }));
    }

    async excluir(id: number) {
        return await this.movRepo.excluir(id);
    }
}

import { NextResponse } from 'next/server';

const service = new UsuarioService(new UsuarioRepository());

        const usuario = await service.buscar(Number(id));
        return NextResponse.json(usuario, { status: 200 });

    } catch (e: any) {
        return NextResponse.json({ erro: e.message }, { status: 400 });
    }
}

        const { id } = await params;

        return NextResponse.json({ erro: e.message }, { status: 400 });

6. Formulário de Produto (Criar / Editar)

app/produtos/form/page.tsx

'use client';
import { useState, useEffect, Suspense } from 'react';
import { useRouter, useSearchParams } from 'next/navigation';
import Link from 'next/link';
import api from '../../lib/api';
import Swal from 'sweetalert2';

function FormularioProdutoContent() {
    const router = useRouter();
    const searchParams = useSearchParams();
    const idParam = searchParams.get('id');

    const [codigo, setCodigo] = useState('');
    const [descricao, setDescricao] = useState('');
    const [quantidade, setQuantidade] = useState('');
    const [valorUnitario, setValorUnitario] = useState('');
    const [estoqueMinimo, setEstoqueMinimo] = useState('');
    const [editandoId, setEditandoId] = useState<number | null>(null);

    useEffect(() => {
        if (idParam) {
            api.get(`/produtos/${idParam}`).then(res => {
                const p = res.data;
                setEditandoId(p.id); setCodigo(p.codigo); setDescricao(p.descricao);
                setQuantidade(String(p.quantidade)); setValorUnitario(String(p.valorUnitario)); setEstoqueMinimo(String(p.estoqueMinimo));
            });
        }
    }, [idParam]);

    const salvar = async (e: React.FormEvent) => {
        e.preventDefault();
        const payload = { codigo, descricao, quantidade: Number(quantidade), valorUnitario: Number(valorUnitario), estoqueMinimo: Number(estoqueMinimo) };
        try {
            if (editandoId) await api.put(`/produtos/${editandoId}`, payload);
            else await api.post('/produtos', payload);
            await Swal.fire('Sucesso!', 'Produto guardado com sucesso.', 'success');
            router.push('/produtos');
        } catch (error: any) {
            Swal.fire('Erro!', error.response?.data?.erro || 'Erro ao guardar.', 'warning');
        }
    };

    return (
        <div className="max-w-2xl mx-auto p-8">
            <div className="flex justify-between items-center mb-6">
                <h1 className="text-2xl font-bold text-slate-800">{editandoId ? 'Editar Produto' : 'Novo Produto'}</h1>
                <Link href="/produtos" className="text-slate-500 hover:underline">Cancelar</Link>
            </div>
            <form onSubmit={salvar} className="bg-white p-6 rounded-lg shadow border space-y-4">
                <div>
                    <label className="block text-sm font-medium text-slate-700 mb-1">Código do Produto</label>
                    <input type="text" required value={codigo} onChange={e => setCodigo(e.target.value)} className="w-full p-2 border rounded outline-none focus:ring-2 focus:ring-blue-500" />
                </div>
                <div>
                    <label className="block text-sm font-medium text-slate-700 mb-1">Descrição</label>
                    <input type="text" required value={descricao} onChange={e => setDescricao(e.target.value)} className="w-full p-2 border rounded outline-none focus:ring-2 focus:ring-blue-500" />
                </div>
                <div className="grid grid-cols-3 gap-4">
                    <div>
                        <label className="block text-sm font-medium text-slate-700 mb-1">Quantidade</label>
                        <input type="number" required value={quantidade} onChange={e => setQuantidade(e.target.value)} className="w-full p-2 border rounded outline-none focus:ring-2 focus:ring-blue-500" />
                    </div>
                    <div>
                        <label className="block text-sm font-medium text-slate-700 mb-1">Preço Unitário</label>
                        <input type="number" step="0.01" required value={valorUnitario} onChange={e => setValorUnitario(e.target.value)} className="w-full p-2 border rounded outline-none focus:ring-2 focus:ring-blue-500" />
                    </div>
                    <div>
                        <label className="block text-sm font-medium text-slate-700 mb-1">Stock Mínimo</label>
                        <input type="number" required value={estoqueMinimo} onChange={e => setEstoqueMinimo(e.target.value)} className="w-full p-2 border rounded outline-none focus:ring-2 focus:ring-blue-500" />
                    </div>
                </div>
                <button type="submit" className="w-full mt-4 bg-blue-600 text-white p-2 rounded font-medium hover:bg-blue-700 transition">Guardar Produto</button>
            </form>
        </div>
    );
}

export default function FormularioProduto() {
    return (
        <Suspense fallback={<div className="p-8 text-center">A carregar...</div>}>
            <FormularioProdutoContent />
        </Suspense>
    );
}
